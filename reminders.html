/* ============================================ NOVA - Notes JavaScript ============================================ */ let allNotes = []; let searchQuery = ""; document.addEventListener("DOMContentLoaded", () => { loadNotes(); initSearch(); initNewNote(); }); async function loadNotes() { try { const result = await getNotes(); allNotes = result.data; renderNotes(); } catch (err) { console.error("Failed to load notes:", err); } } function renderNotes() { const grid = document.getElementById("notesGrid"); if (!grid) return; let filtered = allNotes; if (searchQuery) { const q = searchQuery.toLowerCase(); filtered = filtered.filter((n) => n.title.toLowerCase().includes(q) || n.content.toLowerCase().includes(q) || n.tags.some((t) => t.toLowerCase().includes(q)) ); } // Sort: pinned first, then by date filtered.sort((a, b) => (b.pinned ? 1 : 0) - (a.pinned ? 1 : 0)); if (filtered.length === 0) { grid.innerHTML = `
📝
No notes found
Create your first note or adjust your search.
`; return; } grid.innerHTML = filtered.map((n) => `
📌
${n.title}
${n.content}
${n.tags.map((t) => `${t}`).join("")}
${formatDate(n.date)}
`).join(""); grid.querySelectorAll("[data-action]").forEach((btn) => { btn.addEventListener("click", (e) => { e.stopPropagation(); const card = e.target.closest(".note-card"); const id = Number(card.dataset.id); const action = btn.dataset.action; if (action === "pin") togglePin(id); if (action === "edit") editNote(id); if (action === "delete") removeNote(id); }); }); } async function togglePin(id) { const note = allNotes.find((n) => n.id === id); if (!note) return; note.pinned = !note.pinned; await updateNote(id, { pinned: note.pinned }); renderNotes(); showToast("info", note.pinned ? "Note pinned" : "Note unpinned", note.title); } async function removeNote(id) { await deleteNote(id); allNotes = allNotes.filter((n) => n.id !== id); renderNotes(); showToast("info", "Note deleted", "The note has been removed."); } function editNote(id) { const note = allNotes.find((n) => n.id === id); if (!note) return; document.getElementById("noteTitle").value = note.title; document.getElementById("noteContent").value = note.content; document.getElementById("noteTags").value = note.tags.join(", "); openModal("noteModal"); document.getElementById("saveNoteBtn").dataset.editId = id; } function initSearch() { const search = document.getElementById("noteSearch"); if (!search) return; search.addEventListener("input", () => { searchQuery = search.value.trim(); renderNotes(); }); } function initNewNote() { const btn = document.getElementById("newNoteBtn"); const saveBtn = document.getElementById("saveNoteBtn"); if (!btn || !saveBtn) return; btn.addEventListener("click", () => { document.getElementById("noteTitle").value = ""; document.getElementById("noteContent").value = ""; document.getElementById("noteTags").value = ""; delete saveBtn.dataset.editId; openModal("noteModal"); }); saveBtn.addEventListener("click", async () => { const title = document.getElementById("noteTitle").value.trim(); const content = document.getElementById("noteContent").value.trim(); const tags = document.getElementById("noteTags").value .split(",").map((t) => t.trim()).filter(Boolean); let valid = true; if (!title) { document.getElementById("noteTitleError").classList.add("show"); valid = false; } if (!content) { document.getElementById("noteContentError").classList.add("show"); valid = false; } if (!valid) return; const editId = saveBtn.dataset.editId; if (editId) { await updateNote(Number(editId), { title, content, tags }); const n = allNotes.find((n) => n.id === Number(editId)); if (n) { n.title = title; n.content = content; n.tags = tags; } showToast("success", "Note updated", title); } else { const result = await createNote({ title, content, tags }); allNotes.unshift(result.data); showToast("success", "Note created", title); } closeModal("noteModal"); renderNotes(); }); } document.addEventListener("click", (e) => { if (e.target.hasAttribute("data-close-modal")) { const modal = e.target.closest(".modal-overlay"); if (modal) modal.classList.remove("active"); document.body.style.overflow = ""; } });
